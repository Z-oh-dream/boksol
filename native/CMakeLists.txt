cmake_minimum_required(VERSION 3.20)
project(boksol LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
set(BUILD_TESTING OFF)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)

set(VBAM "${CMAKE_CURRENT_SOURCE_DIR}/../vbam")
set(MELON "${CMAKE_CURRENT_SOURCE_DIR}/../melonds")
set(ENET "${CMAKE_CURRENT_SOURCE_DIR}/../native-deps/enet")

if(GDS_GBA)
    find_package(ZLIB REQUIRED)
    set(VBAM_VERSION 2.2.3)
    set(VBAM_VERSION_RELEASE TRUE)
    set(VBAM_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/vbam-generated")
    set(ENABLE_LINK ON)
    set(ENABLE_DEBUGGER OFF)
    set(SFML_INCLUDE_DIR "${VBAM}/third_party/sfml/include")
    set(SFML_LIBRARIES gds-sfml)
    set(LIBINTL_INC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    set(ZLIB_LIBRARY ZLIB::ZLIB)
    set(ZLIB_INCLUDE_DIR "${ZLIB_INCLUDE_DIRS}")
    add_subdirectory("${VBAM}/third_party/include/nonstd" nonstd)
    add_subdirectory("${VBAM}/third_party/include/ghc" ghc)
    add_subdirectory("${VBAM}/third_party/include/stb" stb)

    set(SFML_ROOT "${VBAM}/third_party/sfml/src")
    file(GLOB SFML_SOURCES
        "${SFML_ROOT}/SFML/System/*.cpp" "${SFML_ROOT}/SFML/System/Win32/*.cpp"
        "${SFML_ROOT}/SFML/Network/*.cpp" "${SFML_ROOT}/SFML/Network/Win32/*.cpp")
    add_library(gds-sfml STATIC ${SFML_SOURCES})
    target_compile_definitions(gds-sfml PUBLIC SFML_STATIC)
    target_include_directories(gds-sfml PUBLIC "${SFML_INCLUDE_DIR}" PRIVATE "${SFML_ROOT}")
    target_link_libraries(gds-sfml PUBLIC ws2_32 winmm nonstd-lib ghc-filesystem)

    include_directories("${VBAM}/src" "${VBAM_GENERATED_DIR}")
    add_subdirectory("${VBAM}/src/core" vbam-core)
    target_compile_definitions(vbam-core PRIVATE C_CORE)
    add_library(gds-gba SHARED gba.cpp)
    target_link_libraries(gds-gba PRIVATE vbam-core)
endif()

if(GDS_DS)
    set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
    set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
    set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
    set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
    set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
    set(DIRENT_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include" CACHE PATH "")
    add_subdirectory("${MELON}" melon)
    set_property(TARGET core PROPERTY CXX_STANDARD 20)
    if(MSVC)
        target_compile_options(core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:/Zc:char8_t-> /FI"${CMAKE_CURRENT_SOURCE_DIR}/include/melon-msvc.h")
    endif()
    add_library(gds-enet STATIC
        "${ENET}/callbacks.c" "${ENET}/compress.c" "${ENET}/host.c"
        "${ENET}/list.c" "${ENET}/packet.c" "${ENET}/peer.c"
        "${ENET}/protocol.c" "${ENET}/win32.c")
    target_include_directories(gds-enet PUBLIC "${ENET}/include")
    target_link_libraries(gds-enet PUBLIC ws2_32 winmm)
    add_library(gds-ds SHARED ds.cpp ds-platform.cpp "${MELON}/src/net/LAN.cpp")
    target_include_directories(gds-ds PRIVATE "${MELON}/src/net")
    target_compile_features(gds-ds PRIVATE cxx_std_20)
    target_compile_definitions(gds-ds PRIVATE __WIN32__)
    target_link_libraries(gds-ds PRIVATE core gds-enet)
endif()
